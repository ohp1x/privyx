class BillingClient:
    """Talk to the billing API.

    Args:
        base_url: The API root, such as ``https://billing.acme.example/v1``.
        api_key: When set, overrides the ``BILLING_API_KEY`` environment variable.
        token: An OAuth access token to send instead of the key.
        timeout: Seconds to wait for a response.
    """

    def __init__(
        self,
        base_url: str,
        api_key: str | None = None,
        token: str = "",
        timeout: float = DEFAULT_TIMEOUT,
    ) -> None:
        self._api_key = api_key or os.environ["BILLING_API_KEY"]
        self._token = token
        self._http = httpx.Client(base_url=base_url, timeout=timeout)

    def login(self, username: str, password: str) -> str:
        response = self._http.post(TOKEN_URL, data={"username": username, "password": password})
        response.raise_for_status()
        return response.json()["access_token"]

    def headers(self) -> dict[str, str]:
        return {"Authorization": f"Bearer {self._token or self._api_key}"}
